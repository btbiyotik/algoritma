# Örnek 2: Dizi üzerinde işlemler - toplam, ortalama, en yüksek, en düşük
BAŞLA
TAMSAYIDİZİSİ notlar[6]
TAMSAYI i = 0, n, toplam = 0, maksimum, minimum, ustunde = 0
ONDALIK ortalama
n = DİZİBOYUTU notlar                    # dizinin kaç elemanlı olduğunu öğren
TEKRARLA giris, @a1, @a2, n
@a1: OKU "Not : ", notlar[i]
@a2: i++
maksimum = notlar[0]                      # ilk eleman başlangıç değeri olur
minimum = notlar[0]
i = 0
TEKRARLA analiz, @b1, @b4, n
@b1: toplam += notlar[i]
EĞER notlar[i] > maksimum İSE maksimum = notlar[i]
EĞER notlar[i] < minimum İSE minimum = notlar[i]
@b4: i++
ortalama = ONDALIĞAÇEVİR toplam / n      # tamsayıyı ondalığa çevirip böl
i = 0
TEKRARLA say, @c1, @c2, n
@c1: EĞER notlar[i] > ortalama İSE ustunde++
@c2: i++
YAZ "Toplam : ", toplam, "   Ortalama : ", ortalama
YAZ "En yüksek : ", maksimum, "   En düşük : ", minimum
YAZ "Ortalamanın üstünde olan öğrenci sayısı : ", ustunde
BİTİR

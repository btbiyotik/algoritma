# Örnek 7: Metin dizisi - isimleri A-Z veya Z-A sıralama, baş harfi yazdırma
BAŞLA
METİNDİZİSİ isimler[5]
TAMSAYI i = 0, secim
TEKRARLA giris, @a1, @a2, 5
@a1: OKU "İsim : ", isimler[i]
@a2: i++
OKU "A'dan Z'ye için 1, Z'den A'ya için 2 : ", secim
DİZİSIRALA isimler, secim == 1           # şart doğruysa A-Z, değilse Z-A
i = 0
TEKRARLA yazdir, @b1, @b2, 5
@b1: YAZ isimler[i], "   (baş harf : ", isimler[i][0], ")"
@b2: i++
BİTİR

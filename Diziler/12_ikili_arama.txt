# Örnek 12: İkili (binary) arama - sıralı dizide aralığı her adımda yarıya indirme
BAŞLA
TAMSAYIDİZİSİ d[10]
TAMSAYI i = 0, alt = 0, ust, orta, aranan, adim = 0, konum = -1
TEKRARLA giris, @a1, @a2, 10
@a1: OKU "Sayı : ", d[i]
@a2: i++
DİZİSIRALA d, DOĞRU                      # ikili arama için dizi SIRALI olmak zorundadır
YAZ "Sıralanmış dizi :"
DİZİYAZDIR d
OKU "Aranan sayı : ", aranan
ust = DİZİBOYUTU d - 1
İKEN ikili, @c1, @c5, alt <= ust         # aralık boşalana kadar
@c1: adim++
orta = (alt + ust) // 2                  # aralığın ortasındaki indeks
YAZ adim, ". adım: aralık [", alt, "-", ust, "], orta indeks ", orta, ", değer ", d[orta]
EĞER d[orta] != aranan İSE GİT @c5
konum = orta
DÖNGÜYÜKIR ikili                         # bulundu
@c5: EĞER d[orta] < aranan İSE alt = orta + 1 DEĞİLSE ust = orta - 1   # yarısını ele
EĞER konum == -1 İSE YAZ aranan, " dizide yok. (", adim, " adımda anlaşıldı)" DEĞİLSE YAZ aranan, " ", konum, ". indekste bulundu. (", adim, " adımda)"
BİTİR
